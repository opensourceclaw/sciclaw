## 4. 类型设计

### 4.1 统一类型定义

```typescript
// src/types/index.ts

export interface ResearchQuery {
  id: string;
  topic: string;
  depth: 'brief' | 'standard' | 'deep';
  sources?: string[];
  language?: 'en' | 'zh';
}

export interface ResearchResult {
  id: string;
  query: ResearchQuery;
  sections: Section[];
  sources: Source[];
  metadata: ResearchMetadata;
  createdAt: Date;
}

export interface Section {
  title: string;
  content: string;
  sources: string[];
  subsections?: Section[];
}

export interface Source {
  url: string;
  title: string;
  authority: number;
  accessedAt: Date;
}

export interface ResearchMetadata {
  duration: number;
  tokensUsed: number;
  searchCount: number;
}
```
