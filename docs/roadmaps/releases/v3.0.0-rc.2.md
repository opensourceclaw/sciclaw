### Personalization Interface

```typescript
interface UserProfile {
  id: string;
  preferences: {
    depth: 'shallow' | 'medium' | 'deep';
    style: 'academic' | 'business' | 'technical' | 'quick';
    format: 'markdown' | 'pdf' | 'html';
    language: string;
  };
  interests: TopicInterest[];
  history: ResearchHistory;
}

interface TopicInterest {
  topic: string;
  weight: number;
  lastResearched: Date;
  frequency: number;
}

// Usage
const profile = await loadUserProfile(userId);
const research = await conductResearch(topic, profile.preferences);
```
