### Reasoning Chain Interface

```typescript
interface ReasoningStep {
  id: string;
  type: 'observation' | 'inference' | 'deduction' | 'conclusion';
  content: string;
  evidence: string[];
  confidence: number;
  next: string[];  // next step IDs
}

interface ReasoningChain {
  id: string;
  topic: string;
  steps: ReasoningStep[];
  conclusion: string;
  confidence: number;
  traceable: boolean;
}

// Usage
const chain = await reason(topic, sources);
// chain.steps, chain.conclusion, chain.confidence
```

### Hypothesis Interface

```typescript
interface Hypothesis {
  id: string;
  statement: string;
  confidence: number;
  supporting: Evidence[];
  opposing: Evidence[];
  testable: boolean;
  testMethod?: string;
}

// Usage
const hypotheses = await generateHypotheses(topic, researchData);
// hypotheses ranked by confidence
```
