### Interactive Research Interface

```typescript
interface InteractiveSession {
  id: string;
  topic: string;
  state: 'exploring' | 'focusing' | 'synthesizing' | 'complete';
  feedback: UserFeedback[];
  direction: ResearchDirection;
  progress: number;  // 0-100
}

interface UserFeedback {
  type: 'steer' | 'approve' | 'reject' | 'refine';
  target: string;  // step or section ID
  content: string;
  timestamp: Date;
}

// Usage
const session = await startInteractiveResearch(topic);
session.onFeedback(async (feedback) => {
  await session.adapt(feedback);
});
```
