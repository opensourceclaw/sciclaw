### Multi-Modal Interface

```typescript
interface MultiModalInput {
  text?: string;
  images?: ImageInput[];
  tables?: TableInput[];
  charts?: ChartInput[];
  pdfs?: PDFInput[];
}

interface ImageInput {
  url: string;
  description?: string;
  analysisType: 'ocr' | 'object' | 'scene' | 'diagram';
}

interface TableInput {
  data: string[][];
  headers?: string[];
  analysisType: 'statistical' | 'comparative' | 'trend';
}

// Usage
const result = await conductResearch({
  topic: 'Market Analysis',
  multimodal: {
    images: [{ url: 'chart.png', analysisType: 'diagram' }],
    tables: [{ data: salesData, analysisType: 'trend' }],
  },
});
```

### Benchmark Interface

interface Benchmark {
