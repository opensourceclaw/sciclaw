```typescript
export interface ReasoningStep {
  id: string;               // crypto.randomUUID()

  evidence: string[];
}

  id: string;
  nodes: CausalNode[];
  edges: CausalEdge[];

  leafEffects: string[];    // node IDs with no outgoing edges
}

  nodes: string[];          // ordered node IDs from cause to effect
  totalStrength: number;    // product of edge strengths along path
  length: number;           // number of edges
}

export interface ReasoningConfig {
  maxSteps: number;         // default: 7
  minConfidence: number;    // default: 0.3
  maxAlternatives: number;  // default: 3
  direction: InferenceDirection; // default: FORWARD
}

  maxSteps: 7,
  minConfidence: 0.3,
  maxAlternatives: 3,
  direction: InferenceDirection.FORWARD,
};
```

### Edge Cases

- `maxSteps` clamped to `[3, 20]`
- `minConfidence` clamped to `[0.1, 0.9]`
- `maxAlternatives` clamped to `[1, 10]`
- `ReasoningStep.dependsOn` must reference existing step IDs within the same chain
- Circular dependencies in `dependsOn` → throw `Error` on chain validation
- Empty `evidence` array → step confidence defaults to `minConfidence`

### Core Algorithm: `executeChain`

```
Input:  ReasoningChain, context
Output: InferenceResult

1. Validate chain structure (no circular deps, max steps)
2. Sort steps topologically by dependsOn
3. For each step (in order):
   a. Compute step confidence from evidence strength:
      - Each evidence item contributes 0.15 to confidence (max 0.9 from evidence)
      - Dependencies' confidence averaged and weighted 0.3
      - Final: clamp(min(evidenceScore + depScore, 1.0), minConfidence, 1.0)
   b. If step.confidence < minConfidence, mark as weak but continue
4. Extract final CONCLUSION step as primary result
5. Collect all CONCLUSION and HYPOTHESIS steps as alternatives
6. Compute overall confidence: weighted average of all step confidences
7. Return InferenceResult
```

### Step Type Sequence Constraints

The chain must follow this order (enforced by `validateChain`):

```
OBSERVATION → HYPOTHESIS → INFERENCE → VALIDATION → CONCLUSION
```

- At least 1 OBSERVATION required
- At least 1 CONCLUSION required
- Minimum 5 steps total (meets acceptance criteria)
- HYPOTHESIS must follow OBSERVATION
- CONCLUSION must be the last step

### Confidence Computation

```typescript
function computeStepConfidence(
  step: ReasoningStep,
  dependencySteps: ReasoningStep[],
  config: ReasoningConfig,
): number {
  // Evidence contribution: 0.15 per evidence item, max 0.9
  const evidenceScore = Math.min(step.evidence.length * 0.15, 0.9);

  // Dependency contribution: average of dep confidences * 0.3
  const depScore = dependencySteps.length > 0
    ? (dependencySteps.reduce((s, d) => s + d.confidence, 0) / dependencySteps.length) * 0.3
    : 0;

  return Math.max(
    Math.min(evidenceScore + depScore, 1.0),
    config.minConfidence,
  );
}
```

### Chain Validation Rules

```typescript
function validateChain(chain: ReasoningChain): { valid: boolean; errors: string[] } {
  const errors: string[] = [];

  // Rule 1: Minimum step count (5)
  if (chain.steps.length < 5) {
    errors.push(`Chain must have at least 5 steps, got ${chain.steps.length}`);
  }

  // Rule 2: Must have OBSERVATION
  if (!chain.steps.some(s => s.type === ReasoningStepType.OBSERVATION)) {
    errors.push("Chain must contain at least one OBSERVATION step");
  }

  // Rule 3: Must have CONCLUSION
  if (!chain.steps.some(s => s.type === ReasoningStepType.CONCLUSION)) {
    errors.push("Chain must contain at least one CONCLUSION step");
  }

  // Rule 4: CONCLUSION must be last
  const lastStep = chain.steps[chain.steps.length - 1];
  if (lastStep && lastStep.type !== ReasoningStepType.CONCLUSION) {
    errors.push("Last step must be of type CONCLUSION");
  }

  // Rule 5: No circular dependencies
  if (hasCycle(chain.steps)) {
    errors.push("Chain contains circular dependencies");
  }

  // Rule 6: All dependsOn references must exist
  const stepIds = new Set(chain.steps.map(s => s.id));
  for (const step of chain.steps) {
    for (const depId of step.dependsOn) {
      if (!stepIds.has(depId)) {
        errors.push(`Step ${step.id} depends on unknown step ${depId}`);
      }
    }
  }

  return { valid: errors.length === 0, errors };
}
```

### Cycle Detection (DFS)

```typescript
function hasCycle(steps: ReasoningStep[]): boolean {
  const stepMap = new Map(steps.map(s => [s.id, s]));
  const visited = new Set<string>();
  const inStack = new Set<string>();

  function dfs(stepId: string): boolean {
    if (inStack.has(stepId)) return true;
    if (visited.has(stepId)) return false;
    visited.add(stepId);
    inStack.add(stepId);
    const step = stepMap.get(stepId);
    if (step) {
      for (const depId of step.dependsOn) {
        if (dfs(depId)) return true;
      }
    }
    inStack.delete(stepId);
    return false;
  }

  for (const step of steps) {
    if (dfs(step.id)) return true;
  }
  return false;
}
```

### Error Handling

- Chain with < 5 steps → `executeChain` returns result with `confidence: 0` and error message
- Circular dependencies → `validateChain` returns `{ valid: false }`, `executeChain` throws
- Missing context keys → step statement treated as-is, no variable interpolation failure
- Empty chain (no steps) → `executeChain` returns empty result with `confidence: 0`

### Edge Cases

- Steps beyond `maxSteps` → `addStep` throws `Error("Max steps exceeded")`
- Duplicate step IDs → `addStep` generates unique ID, no collision
- Very long step statement (>1000 chars) → truncated to 1000 in summary
- All steps at minConfidence → overall confidence = minConfidence

### Core Algorithm: `analyzeCausal`

```
Input:  entities with name and relations
Output: CausalGraph

1. Create CausalNode for each entity
2. Classify each node type:
   - "cause": has outgoing but no incoming edges
   - "effect": has incoming but no outgoing edges
   - "mediator": has both incoming and outgoing edges
   - "confounder": has >2 outgoing edges to nodes that share an effect
3. Create CausalEdge for each relation:
   - Map relation type string → CausalRelationType:
     "causes"/"leads_to"/"results_in" → CAUSES
     "prevents"/"blocks"/"inhibits" → PREVENTS
     "enables"/"allows"/"facilitates" → ENABLES
     "inhibits"/"suppresses"/"reduces" → INHIBITS
     "correlates"/"associated"/"linked" → CORRELATES_WITH
   - Default strength: 0.7 for explicit relations, 0.5 for implicit
4. Identify rootCauses (nodes with no incoming edges)
5. Identify leafEffects (nodes with no outgoing edges)
6. Return CausalGraph
```

### Path Finding Algorithm (BFS)

```typescript
function findPaths(graph: CausalGraph, fromId: string, toId: string): CausalPath[] {
  const paths: CausalPath[] = [];
  const adjacency = buildAdjacency(graph);
  const queue: Array<{ nodeId: string; path: string[]; strength: number }> = [
    { nodeId: fromId, path: [fromId], strength: 1.0 },
  ];

  while (queue.length > 0) {
    const current = queue.shift()!;
    if (current.nodeId === toId) {
      paths.push({
        nodes: current.path,
        totalStrength: current.strength,
        length: current.path.length - 1,
      });
      continue;
    }
    const neighbors = adjacency.get(current.nodeId) ?? [];
    for (const [neighbor, edge] of neighbors) {
      if (!current.path.includes(neighbor)) { // prevent cycles
        queue.push({
          nodeId: neighbor,
          path: [...current.path, neighbor],
          strength: current.strength * edge.strength,
        });
      }
    }
  }

  return paths.sort((a, b) => b.totalStrength - a.totalStrength);
}
```

### Root Cause Identification

```typescript
function findRootCauses(graph: CausalGraph): CausalNode[] {
  const targetIds = new Set(graph.edges.map(e => e.target));
  return graph.nodes.filter(n => !targetIds.has(n.id));
}
```

### Node Classification

```typescript
function classifyNode(
  nodeId: string,
  graph: CausalGraph,
): "cause" | "effect" | "mediator" | "confounder" {
  const outgoing = graph.edges.filter(e => e.source === nodeId);
  const incoming = graph.edges.filter(e => e.target === nodeId);

  if (incoming.length === 0 && outgoing.length > 0) return "cause";
  if (outgoing.length === 0 && incoming.length > 0) return "effect";
  if (incoming.length > 0 && outgoing.length > 0) {
    // Check if confounder: >2 outgoing to nodes sharing a common effect
    if (outgoing.length > 2) {
      const targets = outgoing.map(e => e.target);
      const sharedEffect = graph.nodes.some(n =>
        graph.edges.filter(e => e.target === n.id && targets.includes(e.source)).length > 1
      );
      if (sharedEffect) return "confounder";
    }
    return "mediator";
  }
  return "cause"; // isolated node defaults to cause
}
```

### Error Handling

- Empty entities array → return empty CausalGraph with empty nodes/edges
- Unknown relation type → default to CORRELATES_WITH with strength 0.3
- Self-referencing relation (entity → itself) → skipped with warning
- Duplicate entity names → merge relations, keep first occurrence's ID
- No path between nodes → `findPaths` returns empty array
- Disconnected subgraphs → each subgraph analyzed independently

### Edge Cases

- Single entity → graph with one node, no edges, node is both rootCause and leafEffect
- All nodes connected in a cycle → no rootCauses identified, all nodes are mediators
- Very large graph (>100 nodes) → BFS limited to max 1000 paths to prevent combinatorial explosion
- Edge strength 0 → treated as no relation, excluded from path finding

```typescript
export * from "./types.js";
export * from "./chain.js";
export * from "./causal.js";

export interface ReasoningEngineConfig {
  chain?: Partial<ReasoningConfig>;
  causal?: Partial<ReasoningConfig>;
}

export class ReasoningEngine {
  private chainManager: ReasoningChainManager;
  private causalAnalyzer: CausalAnalyzer;

  constructor(config?: ReasoningEngineConfig);

  // Chain operations (delegated)
  createChain(topic: string, goal: string): ReasoningChain;
  infer(chain: ReasoningChain, context: Record<string, unknown>): InferenceResult;

  // Causal operations (delegated)
  analyzeCausal(entities: Array<{ name: string; relations: Array<{ target: string; type: string }> }>): CausalGraph;

  // Combined operations
  inferWithCausalSupport(
    chain: ReasoningChain,
    entities: Array<{ name: string; relations: Array<{ target: string; type: string }> }>,
    context: Record<string, unknown>,
  ): { inference: InferenceResult; causalGraph: CausalGraph };
}

export function createReasoningEngine(
  config?: ReasoningEngineConfig,
): ReasoningEngine;
```

### Combined Operation: `inferWithCausalSupport`

```
1. Run causal analysis on entities → CausalGraph
2. For each causal path found, add as evidence to relevant chain steps
3. Execute chain with enhanced evidence
4. Return both inference result and causal graph
```

This enables the acceptance criteria: "CausalAnalysis 能识别因果关系" by integrating causal findings into the reasoning chain.
