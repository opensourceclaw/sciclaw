### 1.1 DeepResearchFlow Class

**File**: `src/flows/deep_research_flow.ts`

**Requirements**:
```typescript
export class DeepResearchFlow {
  private currentStage: ResearchStage;
  private researchPlan: ResearchPlan | null;
  private searchResults: SearchResult[] | null;
  private analysisResult: AnalysisResult | null;
  private synthesisResult: SynthesisResult | null;
  
  constructor(config?: DeepResearchConfig);
  
  // Main flow methods
  async start(topic: string): Promise<void>;
  async plan(): Promise<ResearchPlan>;
  async search(): Promise<SearchResult[]>;
  async analyze(): Promise<AnalysisResult>;
  async synthesize(): Promise<SynthesisResult>;
  async report(): Promise<ResearchReport>;
  
  // Gate methods
  async requestApproval(step: string): Promise<boolean>;
  private async waitForApproval(step: string): Promise<void>;
  
  // State management
  getCurrentStage(): ResearchStage;
  getProgress(): number;
}

type ResearchStage = 'plan' | 'search' | 'analyze' | 'synthesize' | 'report';

interface DeepResearchConfig {
  maxDepth?: number;
  timeout?: number;
  approvalRequired?: boolean;
}
```

**Implementation Notes**:
- 每个阶段完成后调用 `waitForApproval()` 等待人工审批
- 支持暂停和恢复
- 记录每个阶段的执行时间

  .option('--mode <mode>', 'Research mode: deep | auto', 'deep')

  .option('--timeout <seconds>', 'Research timeout in seconds', '300')

### 1.4 Gate Strategies

**File**: `src/gate/gate-strategy.ts`

**Requirements**:
```typescript
export interface GateStrategy {
  name: string;
  shouldApprove(stage: string): boolean;
  execute(stage: string): Promise<boolean>;
}

export class DeepResearchGateStrategy implements GateStrategy {
  name = 'deep-research';
  
  shouldApprove(stage: string): boolean {
    // Always require human approval
    return false;
  }
  
  async execute(stage: string): Promise<boolean> {
    // Prompt user for approval
    const answer = await promptUser(`Approve ${stage}?`);
    return answer === 'yes';
  }
}

export class AutoResearchGateStrategy implements GateStrategy {
  name = 'auto-research';
  
  shouldApprove(stage: string): boolean {
    // Auto-approve all
    return true;
  }
  
  async execute(stage: string): Promise<boolean> {
    // Log and auto-approve
    logger.info(`Auto-approved stage: ${stage}`);
    return true;
  }
}
```
